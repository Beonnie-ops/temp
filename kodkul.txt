Функция глОкр(Число, ПорядокОкругления, ОкруглятьВБольшуюСторону=ложь, ВсегдаМеньше=ложь) Экспорт
    
	Перем Результат;

	// Преобразуем порядок округления числа.
	// Если передали пустое значение порядка, то округлим до копеек. 
	Если НЕ ЗначениеЗаполнено(ПорядокОкругления) Тогда
		ПорядокОкругления = ПараметрыСеанса.ПорядокОкругления; 
	КонецЕсли;

	Порядок = Число(Строка(ПорядокОкругления));

	// вычислим количество интервалов, входящих в число
	КоличествоИнтервал = Число / Порядок;

	// вычислим целое количество интервалов.
	КоличествоЦелыхИнтервалов = Цел(КоличествоИнтервал);

	Если КоличествоИнтервал = КоличествоЦелыхИнтервалов Тогда
		
		// Числа поделились нацело. Округлять не нужно.
		Результат = Число;
	Иначе
		Если ОкруглятьВБольшуюСторону Тогда

			// При порядке округления "0.05" 0.371 должно округлитья до 0.4
			Результат = Порядок * (КоличествоЦелыхИнтервалов + 1);
		Иначе

			// При порядке округления "0.05" 0.371 должно округлитья до 0.35,
			// а 0.376 до 0.4
			Результат = Порядок * Окр(КоличествоИнтервал,0,РежимОкругления.Окр15как20);
		КонецЕсли;

	КонецЕсли;
	
	
	Если ВсегдаМеньше Тогда
		Если ПорядокОкругления = Перечисления.ПорядкиОкругления.Окр10 Тогда
			ЧислоОкругления = 10;
		ИначеЕсли ПорядокОкругления = Перечисления.ПорядкиОкругления.Окр50 Тогда
			ЧислоОкругления = 50;
		ИначеЕсли ПорядокОкругления = Перечисления.ПорядкиОкругления.Окр100 Тогда
			ЧислоОкругления = 100;
		ИначеЕсли ПорядокОкругления = Перечисления.ПорядкиОкругления.Окр1000 Тогда
			ЧислоОкругления = 1000;
		ИначеЕсли ПорядокОкругления = Перечисления.ПорядкиОкругления.Окр250 Тогда
			ЧислоОкругления = 250;
		ИначеЕсли ПорядокОкругления = Перечисления.ПорядкиОкругления.Окр500 Тогда
			ЧислоОкругления = 500;	
		Иначе
			ЧислоОкругления = 0;
		КонецЕсли;	
		Если Результат - Число > 0 Тогда
			Результат = Результат - ЧислоОкругления;
		КонецЕсли;	
	КонецЕсли;
	

	Возврат Результат;

Функция глРасчетКалькуляции(ТЗ,ДатаК)  экспорт
	
	Если ТЗ.Количество() = 0 Тогда
		Возврат ТЗ;
	КонецЕсли;
	
	ОсновноеБлюдо = ТЗ[0];
	ССБлюда = Окр(ОсновноеБлюдо.Состав.Итог("Сумма") / ОсновноеБлюдо.РасчетНаКоличество, 2);
	ЦенаБлюдаСНаценкой = ОсновноеБлюдо.Цена;
	
	Для Каждого ТекСтрокаТЗ Из ТЗ Цикл
		СоставТЗ=	ТекСтрокаТЗ.Состав;
		Склад=  ТекСтрокаТЗ.Склад;
		ЦенаБл=ТекСтрокаТЗ.Цена;
		КолБлюд=ТекСтрокаТЗ.Количество;
		КонКолНорм=СоставТЗ.Итог("Сумма");
		Если КонКолНорм = 0 тогда
			Сообщить("Нет цен на продукты в блюде: "+СокрЛП(ТекСтрокаТЗ.Блюдо)+" - Расчет не произведен");
		//	Продолжить;
	    КонецЕсли;	
		РасчетНаКоличество=?(ТекСтрокаТЗ.РасчетНаКоличество=0,1,ТекСтрокаТЗ.РасчетНаКоличество);
		КонКолНорм = ?(КонКолНорм=0,1,КонКолНорм);
		ПроцентНаценкиБл = ?(ЦенаБл=0 ,0,Окр((ЦенаБл/Окр(КонКолНорм/РасчетНаКоличество,4)-1)*100,2));
		Для Каждого ТекСтрокаСоставТЗ Из СоставТЗ Цикл
			если ТипЗНЧ(ТекСтрокаСоставТЗ.Норма) =  Тип("СправочникСсылка.Продукты") тогда
				ТекСтрокаСоставТЗ.Количество= ТекСтрокаСоставТЗ.Брутто*КолБлюд/РасчетНаКоличество;
				ЦенаПродукта = 0;
				ЦенаПродукта=ТекСтрокаСоставТЗ.ЦенаЗаКГ;
				ТекСтрокаСоставТЗ.Сумма=  Окр(ЦенаПродукта*ТекСтрокаСоставТЗ.Количество, 2);

				Если КонКолНорм <> 0  и не ТекСтрокаСоставТЗ.НетЦены тогда
				
					Коэф=ТекСтрокаСоставТЗ.Сумма/КонКолНорм; 
					//ТекСтрокаСоставТЗ.СуммаСНац = Окр(ЦенаБл*Коэф*100,2);
					ТекСтрокаСоставТЗ.СуммаСНац = Окр(ТекСтрокаСоставТЗ.Сумма * ЦенаБлюдаСНаценкой / ССБлюда, 2);
					
					СуммаНац= ТекСтрокаСоставТЗ.СуммаСНац-ТекСтрокаСоставТЗ.Сумма;
				Иначе
				//	ТекСтрокаСоставТЗ.Сумма= 0;
					ТекСтрокаСоставТЗ.СуммаСНац=0;
					СуммаНац=0;
				КонецЕсли;	  
				Если ТекСтрокаСоставТЗ.НетЦены тогда
					ТекСтрокаСоставТЗ.ПроцентНац=0;
					ТекСтрокаСоставТЗ.СуммаНац=0;
				Иначе	
					Если ЦенаПродукта = 0 Тогда
						Сообщить("Нет цены на продукт: "+СокрЛП(ТекСтрокаСоставТЗ.Норма)+"-> Блюдо: "+СокрЛП(ТекСтрокаТЗ.Блюдо)) ;
					КонецЕсли;
					ТекСтрокаСоставТЗ.ПроцентНац =?(ТекСтрокаСоставТЗ.СуммаСНац-СуммаНац=0,0,СуммаНац*100/(ТекСтрокаСоставТЗ.СуммаСНац-СуммаНац)) ;    
					ТекСтрокаСоставТЗ.СуммаНац = ТекСтрокаСоставТЗ.СуммаСНац - ТекСтрокаСоставТЗ.Сумма;
					
				КонецЕсли;	   
			иначе	
				цена=0;
				Если  ТекСтрокаСоставТЗ.Брутто = Null тогда
					Сообщить("Не верно заполнены нормы в блюде: "+СокрЛП(ТекСтрокаТЗ.Блюдо),СтатусСообщения.ОченьВажное) ;
					ТЗ.Очистить();
					Продолжить ;
				КонецЕсли;
				
				ТекСтрокаСоставТЗ.Количество = ТекСтрокаСоставТЗ.Брутто*КолБлюд/РасчетНаКоличество;
				КолРасчет =  КолБлюд*ТекСтрокаСоставТЗ.Брутто;
				РасчетНаКоличествоВход= РасчетНаКоличество* ТекСтрокаСоставТЗ.Норма.РасчетНаКоличество;
				Если  РасчетНаКоличествоВход>=10000 тогда
					#Если Клиент Тогда
					ОчисткаСообщений();
					#КонецЕсли
					Сообщить("Нельзя включать блюдо в блюдо в блюдо: "+СокрЛП(ТекСтрокаСоставТЗ.Норма)+" -> "+СокрЛП(ТекСтрокаТЗ.Блюдо)+" -> ",СтатусСообщения.ОченьВажное) ;
					ТЗ.Очистить();
					Возврат ТЗ;
				КонецЕсли;
				
				глРазложитьБлюдоПоПродуктам(ТекСтрокаСоставТЗ.Норма,ТЗ,цена,КолРасчет,склад,РасчетНаКоличествоВход,датаК,ПроцентНаценкиБл) ; 
				ЦенаПродукта=цена;//
				ТекСтрокаСоставТЗ.ЦенаЗаКГ =ЦенаПродукта;
				ТекСтрокаСоставТЗ.Сумма=Окр(ЦенаПродукта* ТекСтрокаСоставТЗ.Брутто,2);
				Если КонКолНорм <> 0 тогда
					Коэф=ТекСтрокаСоставТЗ.Сумма/КонКолНорм; 
					ТекСтрокаСоставТЗ.СуммаСНац = Окр(ЦенаБл*Коэф*100,2);
					СуммаНац= ТекСтрокаСоставТЗ.СуммаСНац-ТекСтрокаСоставТЗ.Сумма;
					Если		(ТекСтрокаСоставТЗ.СуммаСНац-СуммаНац)=0 тогда
						ТекСтрокаСоставТЗ.ПроцентНац=0;	
						ТекСтрокаСоставТЗ.СуммаНац=0;
					Иначе
						ТекСтрокаСоставТЗ.ПроцентНац =СуммаНац*100/(ТекСтрокаСоставТЗ.СуммаСНац-СуммаНац) ;    
						ТекСтрокаСоставТЗ.СуммаНац = ТекСтрокаСоставТЗ.СуммаСНац - ТекСтрокаСоставТЗ.Сумма;
					КонецЕсли;
				КонецЕсли;
				
				//иначе
				//	Сообщить("Норма "+СокрЛП(ТекСтрокаСоставТЗ.Норма)+"не является ни блюдом, ни продуктом")
			КонецЕсли;
		КонецЦикла;	
	КонецЦикла;	
	Возврат ТЗ;
