Процедура ВывестиОтчет(ТабДок, НомерОтчета) Экспорт
	РасходныеСтроки = Новый Массив();
	ЗарплатныеСтроки = Новый Массив();
	Если НомерОтчета = 1 Тогда
		ТекСКД = ПолучитьМакет("СКДPL");
		ЭтотОбъект.СхемаКомпоновкиДанных = ТекСКД;
		НаборДанных = ПолучитьТзДляОтчета();
		ВывестиВДокумент(ТабДок, ТекСКД, НаборДанных);
		РаскрытьЗапрплатнуюГруппировкуPL(ТабДок);
	ИначеЕсли НомерОтчета = 2 Тогда
		ТекСКД = ПолучитьМакет("СКДCF");
		ЭтотОбъект.СхемаКомпоновкиДанных = ТекСКД;
		НаборДанных = ПолучитьНаборДанныхCF();
		ВывестиВДокумент(ТабДок, ТекСКД, НаборДанных);
		РаскрытьЗапрплатнуюГруппировку(ТабДок);
	КонецЕсли;
КонецПроцедуры

Процедура ВывестиВДокумент(ДокументРезультат, СхемаКомпоновкиДанных, НаборДанных)
	
	//2. Получение настроек компоновки данных
   	Настройки = СхемаКомпоновкиДанных.ВариантыНастроек.Основной.Настройки;

	Если ЭтотОбъект.ВыводитьИтог Тогда
		Настройки.ПараметрыВывода.УстановитьЗначениеПараметра("ГоризонтальноеРасположениеОбщихИтогов", Истина);	
	КонецЕсли;		
	
	//3. Формирование макета компоновки данных на основании схемы и настроек
	КомпоновщикМакета = Новый КомпоновщикМакетаКомпоновкиДанных;
	МакетКомпоновки = КомпоновщикМакета.Выполнить(СхемаКомпоновкиДанных, Настройки);	

	//4. Выполнение компоновки данных
	
	ВнешнийНаборДанных = Новый Структура();
	ВнешнийНаборДанных.Вставить("НаборДанных", НаборДанных);
	ПроцессорКомпоновкиДанных = Новый ПроцессорКомпоновкиДанных;
	ПроцессорКомпоновкиДанных.Инициализировать(МакетКомпоновки, ВнешнийНаборДанных);
	

	//5. Вывод результата компоновки данных в табличный документ
	ДокументРезультат.Очистить();
	ПроцессорВывода = Новый ПроцессорВыводаРезультатаКомпоновкиДанныхВТабличныйДокумент;
	ПроцессорВывода.УстановитьДокумент(ДокументРезультат);
	ПроцессорВывода.Вывести(ПроцессорКомпоновкиДанных);
	
	ДокументРезультат.ПоказатьУровеньГруппировокСтрок(0);
	
КонецПроцедуры

Процедура РаскрытьЗапрплатнуюГруппировкуPL(ДокументРезультат)

		ОбласьЯчеек = ДокументРезультат.Область("R" + Строка(НомерСтроки));
		Если РасходныеСтроки.Найти(ОбласьЯчеек.Текст) <> Неопределено Тогда
			МассивГрупп.Добавить(НомерСтроки);
			
			Если НомерСтроки < МинСтрокаРасх Тогда
				МинСтрокаРасх = НомерСтроки;				
			КонецЕсли;
			
			Если НомерСтроки > МаксСтрокаРасх Тогда
				МаксСтрокаРасх = НомерСтроки;				
			КонецЕсли;
			
		КонецЕсли;
	КонецЦикла;

Функция ПолучитьТзДляОтчета()
	
	Тз = Новый ТаблицаЗначений();
	Тз.Колонки.Вставить(1, "Порядок", Новый ОписаниеТипов("Число"));	
	Тз.Колонки.Вставить(2, "Группа", Новый ОписаниеТипов("Строка",,,, Новый КвалификаторыСтроки(65, ДопустимаяДлина.Фиксированная)));
	Тз.Колонки.Вставить(3, "Дата", Новый ОписаниеТипов("Дата"));
	Тз.Колонки.Вставить(4, "Характеристика", Новый ОписаниеТипов("Строка",,,, Новый КвалификаторыСтроки(40, ДопустимаяДлина.Фиксированная)));
	Тз.Колонки.Вставить(5, "Сумма", Новый ОписаниеТипов("Число"));
	Тз.Колонки.Вставить(0, "УР1");
	Тз.Колонки.Вставить(6, "УР2");
	
	Д1 = НачалоМесяца(ЭтотОбъект.ПериодС);
	Д2 = КонецМесяца(ЭтотОбъект.ПериодПо);

	Пока Д1 < Д2 Цикл
		
		ТЗИтер = ПолучитьНаборДанных(Д1, КонецМесяца(Д1));
		Для Каждого Стр Из ТЗИтер Цикл
			ЗаполнитьЗначенияСвойств(Тз.Добавить(), Стр);	
		КонецЦикла;		
		Сумм = ТЗИтер.ВыгрузитьКолонку("Сумма");
		ИтогСумм = 0;
		Для Каждого СтрСумм Из Сумм Цикл
			ИтогСумм = ИтогСумм + СтрСумм;
		КонецЦикла;
		
		ПрибУб = Тз.Добавить();
		ПрибУб.Порядок = 4;
		ПрибУб.Группа = "";
		ПрибУб.Характеристика = "";
		ПрибУб.Сумма = ИтогСумм;
		ПрибУб.УР2 = "Чистая прибыль";
		ПрибУб.УР1 = "Чистая прибыль";
		ПрибУб.Дата = Д1;
		
		Д1 = ДобавитьМесяц(Д1, 1);
	КонецЦикла;
	

	ПараметрыОтбора = Новый Структура();
	ПараметрыОтбора.Вставить("УР2", "Расходы");
	СтрокиЗП = ТЗ.НайтиСтроки(ПараметрыОтбора);
	Для Каждого СтрЗп Из СтрокиЗП Цикл 
		Если РасходныеСтроки.Найти(СтрЗп.УР1) = Неопределено Тогда
			РасходныеСтроки.Добавить(СтрЗп.УР1);			
		КонецЕсли;
	КонецЦикла;
	
	
	ЗарплатныеСтроки.Добавить("Заработная плата");
	ПараметрыОтбора = Новый Структура();
	ПараметрыОтбора.Вставить("УР1", "Заработная плата");
	СтрокиЗП = ТЗ.НайтиСтроки(ПараметрыОтбора);
	Для Каждого СтрЗп Из СтрокиЗП Цикл 
		Если ЗарплатныеСтроки.Найти(СтрЗп.УР1) = Неопределено Тогда
			ЗарплатныеСтроки.Добавить(СтрЗп.УР1);			
		КонецЕсли;
		
		Если ЗарплатныеСтроки.Найти(СтрЗп.Группа) = Неопределено Тогда
			ЗарплатныеСтроки.Добавить(СтрЗп.Группа);			
		КонецЕсли;

		Если ЗарплатныеСтроки.Найти(СтрЗп.Характеристика) = Неопределено Тогда
			ЗарплатныеСтроки.Добавить(СтрЗп.Характеристика);			
		КонецЕсли;
		
	КонецЦикла;

	
	Возврат Тз;
