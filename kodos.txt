			ДатаПринятия = '00010101'; ДокументПринятия = Неопределено; 
			УправлениеВнеоборотнымиАктивами.ПолучитьДокументБухСостоянияОС(ВыборкаПоОС.ОсновноеСредство, Организация, Перечисления.СостоянияОС.ВведеноВЭксплуатацию, ДокументПринятия, ДатаПринятия);
			
			Если ДатаПринятия = '00010101' Тогда
				ТекстСообщения = "Бух. расчет амортизации: для основного средства " + СокрЛП(ВыборкаПоОС.ОсновноеСредствоПредставление)
				+ " (" + СокрЛП(ВыборкаПоОС.Код) + "), инв. № " + СокрЛП(ВыборкаПоОС.ИнвентарныйНомер);
				ОбщегоНазначения.СообщитьОбОшибке(ТекстСообщения + " не определилась дата начала использования !");
				Продолжить;
			КонецЕсли; 

			ДатаНачалаРасчета = НачалоМесяца(ДобавитьМесяц(ДатаПринятия, 1));


			//ДатаНачалаРасчета = НачалоМесяца(ДобавитьМесяц(ВыборкаПоОС.Период, 1));

			ЧислоЛетПользования = ВыборкаПоОС.СрокИспользованияДляВычисленияАмортизации / 12;
			ЧислоЛетПользования = Цел(ЧислоЛетПользования) + ?(ЧислоЛетПользования - Цел(ЧислоЛетПользования) > 0, 1, 0);
			ЧислоЛетПрошедших   = Цел(ОпределитьФактическийСрокИспользования(ДатаНачалаРасчета, ДатаРасчета) / 12);
			ЧислоЛетОставшихся  = ЧислоЛетПользования - ЧислоЛетПрошедших;

			// Формула суммы чисел от 1 до N
			//		S = (N + 1) * N / 2
			СуммаЧиселЛет = (ЧислоЛетПользования + 1) * ЧислоЛетПользования / 2;
			//ПроцентГодовойАмортизации = ЧислоЛетОставшихся / СуммаЧиселЛет;
            ПроцентГодовойАмортизации = (ЧислоЛетПользования - ЧислоЛетОставшихся + 1) / СуммаЧиселЛет;

			СуммаГодовойАмортизации   = ВыборкаПоОС.СтоимостьДляВычисленияАмортизации * ПроцентГодовойАмортизации;

			Если НЕ ЗначениеЗаполнено(ВыборкаПоОС.ГрафикАмортизации) Тогда
				СуммаАмортизации   = СуммаГодовойАмортизации / 12;
			Иначе
				КоэффициентПериода = ПолучитьКоэффициентАмортизацииИзГрафика(ТаблицаГрафиков, ВыборкаПоОС.ГрафикАмортизации, ДатаРасчета);
				СуммаАмортизации   = СуммаГодовойАмортизации * КоэффициентПериода;
			КонецЕсли;

		КонецЕсли;

		// конечные остатки корректны, т.к. счета учета взяты срезом на конец периода (см. СчетаБУ_ОС_СрезПоследних)
		ВсегоОсталосьСписать = ВыборкаПоОС.СтоимостьКонечныйОстаток - ВыборкаПоОС.АмортизацияКонечныйОстаток;
		СуммаАмортизации     = Макс(0, Мин(СуммаАмортизации, ВсегоОсталосьСписать));

		СтрокаАмортизации = ТаблицаАмортизации.Добавить();

		СтрокаАмортизации.ОС                     = ВыборкаПоОС.ОсновноеСредство;
		СтрокаАмортизации.СчетУчетаБУ            = ВыборкаПоОС.СчетУчета;
		СтрокаАмортизации.СчетАмортизацииБУ      = ВыборкаПоОС.СчетНачисленияАмортизации;
		СтрокаАмортизации.ИмяСубконто            = "ОсновныеСредства";
		СтрокаАмортизации.НаправлениеАмортизации = ВыборкаПоОС.СпособыОтраженияРасходовПоАмортизации;
		СтрокаАмортизации.Бух                    = СуммаАмортизации;
        СтрокаАмортизации.Склад = ВыборкаПоОС.Склад;
				
		СтрокаТаблицаАмортизацииДляОтчета    = ТаблицаАмортизацииДляОтчета.Добавить();
		СтрокаТаблицаАмортизацииДляОтчета.ОС = ВыборкаПоОС.ОсновноеСредство;
		
	КонецЦикла;

	Если  ТаблицаОСДляОтчета.Количество() <> ТаблицаАмортизацииДляОтчета.Количество() Тогда
	
		Сообщить("Есть проблемы , не по всем ОС начислена амортизация");	
		ТЗРазница = РазницаТаблицЗначений(ТаблицаОСДляОтчета,ТаблицаАмортизацииДляОтчета,"ОС");
		
		Если  ТЗРазница.Количество() <> 0 Тогда
					
			Запрос = Новый Запрос;
			Запрос.МенеджерВременныхТаблиц = Новый МенеджерВременныхТаблиц;
			Запрос.Текст = "ВЫБРАТЬ * ПОМЕСТИТЬ ТЗРазница ИЗ &ТЗРазница КАК ТЗРазница";
			Запрос.УстановитьПараметр("ТЗРазница", ТЗРазница);
			Запрос.Выполнить();
						
			Запрос.Текст = 
			"ВЫБРАТЬ
			|	ТЗРазница.ОС.Наименование КАК Наименование,
			|	ТЗРазница.ОС.НомерИнвентарный КАК НомерИнвентарный,
			|	МестонахождениеОСБухгалтерскийУчетСрезПоследних.Местонахождение КАК Подразделение
			|ИЗ
			|	ТЗРазница КАК ТЗРазница
			|		ЛЕВОЕ СОЕДИНЕНИЕ РегистрСведений.МестонахождениеОСБухгалтерскийУчет.СрезПоследних КАК МестонахождениеОСБухгалтерскийУчетСрезПоследних
			|		ПО ТЗРазница.ОС.Ссылка = МестонахождениеОСБухгалтерскийУчетСрезПоследних.ОсновноеСредство
			|ГДЕ
			|	МестонахождениеОСБухгалтерскийУчетСрезПоследних.Организация = &Организация";
			
			Запрос.УстановитьПараметр("Организация",Организация);
			РезультатЗапроса = Запрос.Выполнить().Выгрузить();
			
			ТабДокумент = Новый ТабличныйДокумент;
			Построитель = Новый ПостроительОтчета;
			Построитель.ИсточникДанных=Новый ОписаниеИсточникаДанных(РезультатЗапроса);       
			Построитель.Вывести(ТабДокумент);
			ТабДокумент.Показать("Основные средства по которым не начислена амортизация");    
			  						
		КонецЕсли;  
		
	КонецЕсли;
