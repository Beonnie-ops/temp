// =============================================================================
// Загрузка списка товаров (основных средств) из табличного документа Excel.
//
// В отличие от исходного варианта, где значения читались по ЖЁСТКО заданным
// номерам колонок (Область(НомерСтроки, 1), Область(НомерСтроки, 2) и т.д.),
// здесь нужные колонки сначала ИЩУТСЯ ПО ТЕКСТУ ЗАГОЛОВКА, а уже потом из
// найденных колонок читаются значения. Это нужно, чтобы код продолжал
// корректно работать, даже если в исходном файле колонки будут добавлены,
// удалены или переставлены местами.
//
// Из файла забираются три поля:
//   - Наименование товара
//   - Инвентарный номер
//   - Код ТН ВЭД
//
// Примечание. Оригинальный файл с кодом, который прислал пользователь, был
// повреждён при загрузке (кириллический текст оказался заменён на символы
// "неизвестный символ" — U+FFFD, побайтово, без возможности восстановления
// исходных имён переменных). Поэтому ниже — не построчная правка старого
// файла, а его чистая пересборка на основе видимой структуры (диалог выбора
// файла, чтение ТабличногоДокумента, цикл по строкам начиная с 11-й,
// заполнение результирующей таблицы), адаптированная под задачу:
// получать значения из колонок "Наименование товара", "Инвентарный номер"
// и "Код ТН ВЭД".
// =============================================================================

Функция ЗагрузитьТоварыИзEXCEL(ИмяФайлаПоУмолчанию = "") Экспорт
	
	ИмяФайла = ВыбратьФайлEXCEL(ИмяФайлаПоУмолчанию);
	Если ПустаяСтрока(ИмяФайла) Тогда
		Возврат Неопределено;
	КонецЕсли;
	
	ТабДок = Новый ТабличныйДокумент;
	
	Попытка
		ТабДок.Прочитать(ИмяФайла, СпособЧтенияЗначенийТабличногоДокумента.Значение);
	Исключение
		Предупреждение("Не удалось прочитать файл """ + ИмяФайла + """: " + ОписаниеОшибки());
		Возврат Неопределено;
	КонецПопытки;
	
	Результат = НоваяТаблицаТоваров();
	
	// --- 1. Ищем строку заголовков и номера нужных колонок --------------------
	
	СтруктураКолонок = НайтиКолонкиТаблицы(ТабДок);
	
	Если СтруктураКолонок = Неопределено Тогда
		Предупреждение(
			"В файле """ + ИмяФайла + """ не удалось найти строку заголовков с колонками
			|""Наименование товара"", ""Инвентарный номер"" и ""Код ТН ВЭД"".
			|Проверьте, что заголовки присутствуют в файле, либо скорректируйте
			|константу МаксимальнаяСтрокаПоискаЗаголовков в функции НайтиКолонкиТаблицы.");
		Возврат Неопределено;
	КонецЕсли;
	
	НомерСтрокиЗаголовков       = СтруктураКолонок.НомерСтроки;
	НомерКолонкиНаименование    = СтруктураКолонок.Наименование;
	НомерКолонкиИнвентарныйНомер = СтруктураКолонок.ИнвентарныйНомер;
	НомерКолонкиТНВЭД           = СтруктураКолонок.ТНВЭД;
	
	// --- 2. Читаем строки данных, начиная со строки, следующей за заголовком ---
	
	КоличествоСтрок = ТабДок.ВысотаТаблицы;
	НомерСтроки     = НомерСтрокиЗаголовков + 1;
	
	Пока НомерСтроки <= КоличествоСтрок Цикл
		
		Наименование = "";
		Если НомерКолонкиНаименование <> 0 Тогда
			Наименование = СокрЛП(ТабДок.Область(НомерСтроки, НомерКолонкиНаименование).Текст);
		КонецЕсли;
		
		ИнвентарныйНомер = "";
		Если НомерКолонкиИнвентарныйНомер <> 0 Тогда
			ИнвентарныйНомер = СокрЛП(ТабДок.Область(НомерСтроки, НомерКолонкиИнвентарныйНомер).Текст);
		КонецЕсли;
		
		КодТНВЭД = "";
		Если НомерКолонкиТНВЭД <> 0 Тогда
			КодТНВЭД = СокрЛП(ТабДок.Область(НомерСтроки, НомерКолонкиТНВЭД).Текст);
		КонецЕсли;
		
		// Полностью пустую строку (например, разделитель разделов таблицы) пропускаем.
		Если Наименование = "" И ИнвентарныйНомер = "" И КодТНВЭД = "" Тогда
			НомерСтроки = НомерСтроки + 1;
			Продолжить;
		КонецЕсли;
		
		НоваяСтрока = Результат.Добавить();
		НоваяСтрока.Наименование      = Наименование;
		НоваяСтрока.ИнвентарныйНомер  = ИнвентарныйНомер;
		НоваяСтрока.ТНВЭД             = КодТНВЭД;
		НоваяСтрока.НомерСтрокиИсходногоФайла = НомерСтроки;
		
		НомерСтроки = НомерСтроки + 1;
		
	КонецЦикла;
	
	Возврат Результат;
	
КонецФункции

// Формирует пустую таблицу значений с колонками, которые заполняются
// функцией ЗагрузитьТоварыИзEXCEL.
Функция НоваяТаблицаТоваров()
	
	ТЗ = Новый ТаблицаЗначений;
	ТЗ.Колонки.Добавить("Наименование");
	ТЗ.Колонки.Добавить("ИнвентарныйНомер");
	ТЗ.Колонки.Добавить("ТНВЭД");
	ТЗ.Колонки.Добавить("НомерСтрокиИсходногоФайла");
	Возврат ТЗ;
	
КонецФункции

// Просматривает первые несколько строк табличного документа в поисках строки
// заголовков, содержащей колонки "Наименование товара", "Инвентарный номер"
// и "Код ТН ВЭД" (поиск нестрогий, по вхождению подстроки, без учёта регистра).
// Возвращает структуру {НомерСтроки, Наименование, ИнвентарныйНомер, ТНВЭД}
// с номерами найденных колонок (0, если колонка не найдена), либо Неопределено,
// если строку заголовков найти не удалось вообще.
Функция НайтиКолонкиТаблицы(ТабДок)
	
	МаксимальнаяСтрокаПоискаЗаголовков = 20;
	
	Для НомерСтроки = 1 По Мин(МаксимальнаяСтрокаПоискаЗаголовков, ТабДок.ВысотаТаблицы) Цикл
		
		НомерКолонкиНаименование     = НайтиНомерКолонкиПоЗаголовку(ТабДок, НомерСтроки, "наименование");
		НомерКолонкиИнвентарныйНомер = НайтиНомерКолонкиПоЗаголовку(ТабДок, НомерСтроки, "инвентарн");
		НомерКолонкиТНВЭД            = НайтиНомерКолонкиПоЗаголовку(ТабДок, НомерСтроки, "тн вэд");
		Если НомерКолонкиТНВЭД = 0 Тогда
			НомерКолонкиТНВЭД = НайтиНомерКолонкиПоЗаголовку(ТабДок, НомерСтроки, "тнвэд");
		КонецЕсли;
		
		// Строку считаем строкой заголовков, если нашли хотя бы наименование
		// и ещё одну из двух оставшихся колонок.
		Если НомерКолонкиНаименование <> 0
			И (НомерКолонкиИнвентарныйНомер <> 0 Или НомерКолонкиТНВЭД <> 0) Тогда
			
			Результат = Новый Структура;
			Результат.Вставить("НомерСтроки", НомерСтроки);
			Результат.Вставить("Наименование", НомерКолонкиНаименование);
			Результат.Вставить("ИнвентарныйНомер", НомерКолонкиИнвентарныйНомер);
			Результат.Вставить("ТНВЭД", НомерКолонкиТНВЭД);
			Возврат Результат;
			
		КонецЕсли;
		
	КонецЦикла;
	
	Возврат Неопределено;
	
КонецФункции

// Ищет в указанной строке табличного документа колонку, текст заголовка
// которой содержит ИскомыйТекст (без учёта регистра). Возвращает номер
// найденной колонки, либо 0, если колонка не найдена.
Функция НайтиНомерКолонкиПоЗаголовку(ТабДок, НомерСтроки, ИскомыйТекст)
	
	ИскомыйТекстНижнийРегистр = НРег(ИскомыйТекст);
	
	Для НомерКолонки = 1 По ТабДок.ШиринаТаблицы Цикл
		
		ТекстЯчейки = НРег(СокрЛП(ТабДок.Область(НомерСтроки, НомерКолонки).Текст));
		
		Если Найти(ТекстЯчейки, ИскомыйТекстНижнийРегистр) <> 0 Тогда
			Возврат НомерКолонки;
		КонецЕсли;
		
	КонецЦикла;
	
	Возврат 0;
	
КонецФункции

// Показывает диалог выбора файла Excel и возвращает выбранное полное имя файла
// (или пустую строку, если пользователь отменил выбор).
Функция ВыбратьФайлEXCEL(ИмяФайлаПоУмолчанию = "")
	
	ДиалогВыбораФайла = Новый ДиалогВыбораФайла(РежимДиалогаВыбораФайла.Открытие);
	ДиалогВыбораФайла.Заголовок = "Выберите файл со списком товаров для загрузки";
	ДиалогВыбораФайла.Фильтр    = "Файлы Excel|*.xls;*.xlsx";
	ДиалогВыбораФайла.ПолноеИмяФайла = "";
	ДиалогВыбораФайла.Множественныйвыбор = Ложь;
	
	Если Не ПустаяСтрока(ИмяФайлаПоУмолчанию) Тогда
		ДиалогВыбораФайла.ПолноеИмяФайла = ИмяФайлаПоУмолчанию;
	КонецЕсли;
	
	ДиалогВыбораФайла.Каталог = "";
	ДиалогВыбораФайла.Проверка = Истина;
	
	Если ДиалогВыбораФайла.Выбрать() Тогда
		Возврат ДиалогВыбораФайла.ПолноеИмяФайла;
	КонецЕсли;
	
	Возврат "";
	
КонецФункции
